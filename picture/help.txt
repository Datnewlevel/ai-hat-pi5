# Tạo file train
nano train.py

from ultralytics import YOLO

model = YOLO('yolov8s.pt')

model.train(
    data="/home/your_username/yolo_project/dataset/data.yaml",  
    epochs=150,
    imgsz=640,
    batch=16,       # Giảm xuống 8 nếu bị lỗi RAM
    name="my_model",
    device=0        # Dùng GPU, đổi thành "cpu" nếu không có GPU
)

# Chạy file train
python3 train.py

#export to ONNX
yolo export model=runs/detect/my_model/weights/best.pt imgsz=640 format=onnx opset=11
